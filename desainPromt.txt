      class GoldParticle {
        constructor() {
          this.reset();
        }
        reset() {
          this.x = Math.random() * canvas.width;
          this.y = Math.random() * canvas.height;
          this.size = Math.random() * 2.8 + 0.6;
          this.speedY = Math.random() * 0.7 + 0.2;
          this.speedX = (Math.random() - 0.5) * 0.4;
          this.alpha = Math.random() * 0.6 + 0.2;
          this.pulsing = Math.random() * 0.02 + 0.005;
        }
        update() {
          this.y += this.speedY;
          this.x += this.speedX;
          this.alpha += Math.sin(Date.now() * this.pulsing) * 0.01;

          if (this.y > canvas.height) {
            this.y = -10;
            this.x = Math.random() * canvas.width;
          }
        }
        draw() {
          ctx.beginPath();
          ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
          ctx.fillStyle = `rgba(215, 188, 137, ${Math.max(0.1, Math.min(1, this.alpha))})`;
          ctx.shadowBlur = 8;
          ctx.shadowColor = "rgba(231, 212, 178, 0.8)";
          ctx.fill();
        }
      }

          submitBtn.innerText = "Mengirim...";
        }

        try {
          var apiBase = getApiBaseUrl();
          var slug = getWeddingSlug();
          var mappedAttendance =
            status === "Tidak Hadir" ? "not-attending" : "attending";
          var guestPax = status === "Tidak Hadir" ? 0 : 1;

          var res = await fetch(apiBase + "/wishes", {
            method: "POST",
            headers: {
              "Content-Type": "application/json",
            },
            body: JSON.stringify({
              slug: slug,
              name: name,
              attendance: mappedAttendance,
              guests: guestPax,
              message: message,
            }),
          });

          var json = await res.json();
          if (res.ok && json.success) {
            var wishesList = document.getElementById("wishesList");
            if (wishesList) {
              var newCard = createWishCardElement(name, status, message);
              wishesList.prepend(newCard);
            }
            wishesTotal += 1;
            var counterEl = document.getElementById("wishesCounter");
            if (counterEl) counterEl.textContent = wishesTotal + " Pesan";

            showToast(
              "Terima kasih, konfirmasi kehadiran dan doa Anda telah tersimpan!",
            );
            if (messageInput) messageInput.value = "";
          } else {
            showToast(
              json.message || "Gagal mengirim ucapan, coba lagi nanti.",
            );
          }
        } catch (err) {
          console.error("Gagal koneksi endpoint:", err);
          // Fallback UI update jika offline
          var wishesList = document.getElementById("wishesList");
          if (wishesList) {
            var newCard = createWishCardElement(name, status, message);
            wishesList.prepend(newCard);
          }
          showToast("Konfirmasi kehadiran dan doa Anda telah dicatat!");
          if (messageInput) messageInput.value = "";
        } finally {
          if (submitBtn) {
            submitBtn.disabled = false;
            submitBtn.innerText = originalText;
          }
        }
      }

      // Otomatis fetch daftar doa saat dokumen siap
      if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", fetchWishesFeed);
      } else {
        fetchWishesFeed();
      }

      function escapeHtml(text) {
        const map = {
          "&": "&amp;",
          "<": "&lt;",
          ">": "&gt;",
          '"': "&quot;",
          "'": "&#039;",
        };
        return text.replace(/[&<>"']/g, (m) => map[m]);
      }

      function openLightbox(element) {
        const img = element.querySelector("img");
        if (!img) return;

        const overlay = document.createElement("div");
        overlay.className =
          "fixed inset-0 z-50 bg-black/90 flex items-center justify-center p-4 cursor-pointer transition-opacity duration-300 opacity-0";
        overlay.innerHTML = `
        <div class="relative max-w-4xl max-h-[85vh] overflow-hidden rounded-2xl shadow-2xl">
          <img src="${img.src}" class="w-full h-full object-contain max-h-[80vh] rounded-xl">
          <button class="absolute top-4 right-4 text-white bg-black/60 w-10 h-10 rounded-full flex items-center justify-center text-lg font-bold hover:bg-black/80 transition-colors">&times;</button>
        </div>
      `;
        document.body.appendChild(overlay);
        setTimeout(() => overlay.classList.remove("opacity-0"), 20);

        overlay.addEventListener("click", () => {
          overlay.classList.add("opacity-0");
          setTimeout(() => overlay.remove(), 300);
        });
      }
